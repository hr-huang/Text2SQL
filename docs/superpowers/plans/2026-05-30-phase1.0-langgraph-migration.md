# Phase 1.0: LangGraph Migration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate the hand-written serial pipeline in `app/workflow/graph.py` to a proper LangGraph `StateGraph` with declarative routing, preserving all existing behavior.

**Architecture:** Replace the imperative `run_text2sql_workflow()` function with a `StateGraph` that defines 12 nodes and 4 conditional routing functions. Nodes keep their existing `(state) -> dict` signature and are unchanged. The graph compiles once at startup and reuses a `CompiledGraph` for all requests.

**Tech Stack:** LangGraph >= 0.2.0, existing FastAPI + nodes unchanged

**Key Design Decisions:**
- Nodes return **partial state dicts** (only changed fields), which LangGraph merges via overwrite
- Initial state must include ALL TypedDict keys with defaults (LangGraph requirement for `total=False`)
- Three trivial "terminal" nodes are extracted from current inline code: `answer_non_data`, `answer_validation_failed`, `answer_exec_failed`
- Loop `execute → sql_repair → execute` handled via `repair_attempts` counter + conditional edge
- Graph compiled once in `main.py` startup, stored as module-level variable
- Tracing (add_trace) is extracted into a `trace_wrapper` — each node is wrapped to auto-record its output

**Graph structure:**

```
START → intent ──(非数据)──► answer_non_data → END
              │
              └─(数据)────► classify → semantic → schema
                                               │
                                               ▼
                                            sql_gen → validate ──(失败)──► answer_validation_failed → END
                                                             │
                                                             └─(通过)──► execute ──(成功)──► answer → END
                                                                              │
                                                                              └─(失败)──► sql_repair ──(修复成功)──► execute (loop)
                                                                                                │
                                                                                                └─(放弃)──► answer_exec_failed → END
```

```python
# app/workflow/trace_wrapper.py
"""Node wrapper that automatically records trace entries for LangGraph nodes."""

from typing import Any, Callable
from app.schemas.state import Text2SQLState
from app.utils.trace_utils import add_trace


def traced(node_name: str) -> Callable:
    """Decorator that wraps a node function to auto-record its output via add_trace.

    Usage:
        @traced("intent")
        def intent_node(state: Text2SQLState) -> dict:
            ...
    """

    def decorator(fn: Callable[[Text2SQLState], dict]) -> Callable[[Text2SQLState], dict]:
        def wrapper(state: Text2SQLState) -> dict:
            result = fn(state)
            # Only record non-empty outputs
            trace_output = {k: v for k, v in result.items() if v not in (None, "", [], {})}
            if trace_output:
                state = add_trace(state, node=node_name, output=trace_output)
                # Put the updated debug_trace back into the return dict
                result["debug_trace"] = state.get("debug_trace", [])
            return result
        return wrapper

    return decorator
```

```python
# app/workflow/graph.py
"""Text2SQL workflow built on LangGraph StateGraph.

Nodes are regular functions (state) -> dict.
The graph is compiled once at startup and reused for all requests.
"""

from langgraph.graph import StateGraph, START, END

from app.nodes.intent_node import intent_node
from app.nodes.classify_node import classify_node
from app.nodes.semantic_parse_node import semantic_parse_node
from app.nodes.schema_retrieval_node import schema_retrieval_node
from app.nodes.sql_generation_node import sql_generation_node
from app.nodes.sql_validation_node import sql_validation_node
from app.nodes.sql_execution_node import sql_execution_node
from app.nodes.sql_repair_node import sql_repair_node
from app.nodes.answer_node import answer_node
from app.nodes.terminal_nodes import (
    answer_non_data_node,
    answer_validation_failed_node,
    answer_exec_failed_node,
)
from app.schemas.state import Text2SQLState
from app.workflow.trace_wrapper import traced


# ── Routing functions (called by conditional edges) ─────────────────

def route_after_intent(state: Text2SQLState) -> str:
    """After intent detection: proceed to classify or reject non-data questions."""
    if state.get("is_data_question"):
        return "classify"
    return "answer_non_data"


def route_after_validate(state: Text2SQLState) -> str:
    """After SQL validation: execute if valid, otherwise report failure."""
    if state.get("sql_validation_error"):
        return "answer_validation_failed"
    return "execute"


def route_after_execute(state: Text2SQLState) -> str:
    """After SQL execution: answer if success, repair if failed (under limit)."""
    if not state.get("execution_error"):
        return "answer"
    if state.get("repair_attempts", 0) < 3:
        return "sql_repair"
    return "answer_exec_failed"


def route_after_repair(state: Text2SQLState) -> str:
    """After repair attempt: retry execution if fixed, otherwise give up."""
    if state.get("execution_error"):
        return "answer_exec_failed"
    return "execute"


# ── Graph construction ──────────────────────────────────────────────

def compile_graph():
    """Build and compile the Text2SQL StateGraph.

    Called once at application startup. Returns a CompiledGraph that can be
    invoked with `graph.invoke(state)` or streamed with `graph.astream(state)`.
    """
    builder = StateGraph(Text2SQLState)

    # ── Add nodes ──
    # Core pipeline nodes (wrapped with auto-tracing)
    builder.add_node("intent", traced("intent")(intent_node))
    builder.add_node("classify", traced("classify")(classify_node))
    builder.add_node("semantic", traced("semantic")(semantic_parse_node))
    builder.add_node("schema", traced("schema")(schema_retrieval_node))
    builder.add_node("sql_gen", traced("sql_gen")(sql_generation_node))
    builder.add_node("validate", traced("validate")(sql_validation_node))
    builder.add_node("execute", traced("execute")(sql_execution_node))
    builder.add_node("sql_repair", traced("sql_repair")(sql_repair_node))
    builder.add_node("answer", traced("answer")(answer_node))

    # Terminal nodes (early-exit paths)
    builder.add_node("answer_non_data", traced("answer_non_data")(answer_non_data_node))
    builder.add_node("answer_validation_failed", traced("answer_validation_failed")(answer_validation_failed_node))
    builder.add_node("answer_exec_failed", traced("answer_exec_failed")(answer_exec_failed_node))

    # ── Add edges ──

    # Entry
    builder.add_edge(START, "intent")

    # intent → classify  OR  answer_non_data → END
    builder.add_conditional_edges("intent", route_after_intent, {
        "classify": "classify",
        "answer_non_data": "answer_non_data",
    })
    builder.add_edge("answer_non_data", END)

    # classify → semantic → schema → sql_gen → validate (serial chain)
    builder.add_edge("classify", "semantic")
    builder.add_edge("semantic", "schema")
    builder.add_edge("schema", "sql_gen")
    builder.add_edge("sql_gen", "validate")

    # validate → execute  OR  answer_validation_failed → END
    builder.add_conditional_edges("validate", route_after_validate, {
        "execute": "execute",
        "answer_validation_failed": "answer_validation_failed",
    })
    builder.add_edge("answer_validation_failed", END)

    # execute → answer → END  OR  sql_repair  OR  answer_exec_failed → END
    builder.add_conditional_edges("execute", route_after_execute, {
        "answer": "answer",
        "sql_repair": "sql_repair",
        "answer_exec_failed": "answer_exec_failed",
    })
    builder.add_edge("answer", END)

    # sql_repair → execute (retry)  OR  answer_exec_failed (give up)
    builder.add_conditional_edges("sql_repair", route_after_repair, {
        "execute": "execute",
        "answer_exec_failed": "answer_exec_failed",
    })
    builder.add_edge("answer_exec_failed", END)

    return builder.compile()


# ── Convenience: run workflow synchronously ──────────────────────────

# Compiled once at startup (see app/main.py). This module-level variable
# is set after compile_graph() is called.
_graph = None  # type: ignore


def get_graph():
    """Return the compiled graph. Raises RuntimeError if not yet compiled."""
    if _graph is None:
        raise RuntimeError(
            "Graph not compiled yet. Call compile_graph() at startup."
        )
    return _graph


def set_graph(compiled_graph):
    """Store the compiled graph (called from main.py startup)."""
    global _graph
    _graph = compiled_graph


def run_text2sql_workflow(state: Text2SQLState) -> Text2SQLState:
    """Run the Text2SQL workflow synchronously.

    Preserved for backward compatibility with existing callers
    (API route, evaluation script).
    """
    graph = get_graph()
    result = graph.invoke(state)
    # graph.invoke returns the full state — cast to Text2SQLState
    return Text2SQLState(**{k: v for k, v in result.items() if k in Text2SQLState.__optional_keys__ or k in Text2SQLState.__required_keys__})
```

```python
"""评测脚本：运行评测集，计算执行准确率"""
import json
import sqlite3
import time
from pathlib import Path

from app.workflow.state_helpers import create_initial_state
from app.workflow.graph import compile_graph

EVAL_PATH = Path(__file__).resolve().parent.parent / "data" / "eval_questions.json"
DB_PATH   = Path(__file__).resolve().parent.parent / "data" / "chinook.db"
OUT_DIR   = Path(__file__).resolve().parent.parent / "data"


def run_eval_set(questions, graph):
    """跑评测集"""
    results = []
    passed = 0
    total_time = 0

    for q in questions:
        state = create_initial_state(
            user_id="eval",
            question=q["question"],
            datasource_id="chinook_db",
            session_id=f"eval_{q['id']}",
        )

        t0 = time.time()
        try:
            final = graph.invoke(state)
        except Exception as e:
            results.append({
                "id": q["id"],
                "question": q["question"],
                "difficulty": q["difficulty"],
                "status": "error",
                "error": str(e)[:200],
                "generated_sql": "",
                "gold_sql": q["gold_sql"],
                "match": False,
            })
            continue

        elapsed = time.time() - t0
        total_time += elapsed

        generated_sql = final.get("validated_sql") or final.get("generated_sql") or ""
        rows = final.get("execution_result", [])
        has_error = bool(final.get("execution_error"))

        gold_rows = execute_gold_sql(q["gold_sql"])
        match = compare_results(rows, gold_rows)
        complexity = final.get("complexity", "?")

        if match:
            passed += 1

        results.append({
            "id": q["id"],
            "question": q["question"][:60],
            "difficulty": q["difficulty"],
            "status": "pass" if match else ("exec_error" if has_error else "mismatch"),
            "generated_sql": generated_sql[:200],
            "gold_sql": q["gold_sql"][:200],
            "match": match,
            "complexity": complexity,
            "rows_count": len(rows) if rows else 0,
            "gold_rows_count": len(gold_rows) if gold_rows else 0,
            "elapsed": round(elapsed, 2),
        })

    accuracy = passed / len(questions) * 100 if questions else 0
    avg_time = total_time / len(questions) if questions else 0

    return {
        "total": len(questions),
        "passed": passed,
        "failed": len(questions) - passed,
        "accuracy": round(accuracy, 1),
        "avg_time_sec": round(avg_time, 1),
        "results": results,
    }


def execute_gold_sql(sql: str) -> list[dict]:
    """执行标准 SQL，返回结果集"""
    try:
        conn = sqlite3.connect(DB_PATH)
        conn.row_factory = sqlite3.Row
        cursor = conn.execute(sql)
        rows = [dict(row) for row in cursor.fetchall()]
        conn.close()
        return rows
    except Exception:
        return []


def compare_results(generated_rows: list[dict], gold_rows: list[dict]) -> bool:
    """比较两个结果集是否等价"""
    if len(generated_rows) != len(gold_rows):
        return False
    if not generated_rows and not gold_rows:
        return True

    def normalize(rows):
        if not rows:
            return []
        tuples = [tuple(sorted(str(v) for v in r.values())) for r in rows]
        return sorted(tuples)

    return normalize(generated_rows) == normalize(gold_rows)


def print_report(summary, label=""):
    """打印评测报告"""
    print(f"\n{'='*60}")
    print(f"  评测报告 {label}")
    print(f"{'='*60}")
    print(f"  总题数:      {summary['total']}")
    print(f"  通过:        {summary['passed']}")
    print(f"  失败:        {summary['failed']}")
    print(f"  执行准确率:  {summary['accuracy']}%")
    print(f"  平均耗时:    {summary['avg_time_sec']:.1f}s")
    print(f"{'='*60}")

    for diff in ["simple", "medium", "complex"]:
        d_results = [r for r in summary["results"] if r["difficulty"] == diff]
        if not d_results:
            continue
        d_pass = sum(1 for r in d_results if r["match"])
        print(f"  {diff:8s}: {d_pass}/{len(d_results)} ({d_pass/len(d_results)*100:.0f}%)")

    failures = [r for r in summary["results"] if not r["match"]]
    if failures:
        print(f"\n  失败详情:")
        for f in failures:
            print(f"    [{f['id']}] {f['question'][:50]}")
            print(f"      状态: {f['status']}  | 行数: {f['rows_count']}/{f['gold_rows_count']}")
            print(f"      生成: {f['generated_sql'][:120]}")

    complexities = {}
    for r in summary["results"]:
        c = r.get("complexity", "?")
        complexities[c] = complexities.get(c, 0) + 1
    print(f"\n  分类分布: {complexities}")


def main():
    with open(EVAL_PATH, encoding="utf-8") as f:
        data = json.load(f)
    questions = data["questions"]

    print(f"加载评测集: {len(questions)} 题")

if __name__ == "__main__":
```
